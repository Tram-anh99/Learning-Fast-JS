#### File: `giong/gen_caygiong.xlsx`
```

─────────────────────────────────────────────

ten_cay_trong         →  loai_cay.ten_cay
ten_khoa_hoc          →  loai_cay.ten_khoa_hoc
```

**Script Python:**
```python
import pandas as pd
import psycopg2
from psycopg2.extras import execute_batch

# Đọc Excel
df = pd.read_excel('giong/gen_caygiong.xlsx', sheet_name='Sheet2')

# Kết nối PostgreSQL
conn = psycopg2.connect(
    host="localhost",
    database="nongsan_db",
    user="postgres",
    password="your_password"
)
cur = conn.cursor()

# Insert data
for _, row in df.iterrows():
    cur.execute("""
        INSERT INTO nongsan.loai_cay (ma_cay, ten_cay, ten_khoa_hoc)
        VALUES (%s, %s, %s)
        ON CONFLICT (ma_cay) DO NOTHING;
    """, (
        f"CAY_{row['caytrong_ID']}",
        row['ten_cay_trong'],
        row['ten_khoa_hoc']
    ))

conn.commit()
cur.close()
conn.close()
```

**Script Python phức tạp hơn:**
```python
import pandas as pd
import psycopg2

df = pd.read_excel('msvt/msvt_thitruongvungtrong.xlsx')

conn = psycopg2.connect(...)
cur = conn.cursor()

for _, row in df.iterrows():
    # 1. Tìm/tạo tỉnh, huyện, xã
    cur.execute("""
        SELECT id FROM nongsan.tinh WHERE ten_tinh = %s
    """, (row['tinh'],))
    tinh_id = cur.fetchone()
    
    if not tinh_id:
        cur.execute("""
            INSERT INTO nongsan.tinh (ma_tinh, ten_tinh)
            VALUES (%s, %s) RETURNING id
        """, (row['tinh'][:10], row['tinh']))
        tinh_id = cur.fetchone()
    
    # 2. Tạo vùng trồng
    cur.execute("""
        INSERT INTO nongsan.vung_trong 
        (ma_vung, ma_vung_puc, ten_vung, dien_tich, tinh_id)
        VALUES (%s, %s, %s, %s, %s)
        RETURNING id
    """, (
        row['maso'] if pd.notna(row['maso']) else row['mavungtrong_puc'],
        row['mavungtrong_puc'],
        row['tenvungtrong'],
        row['dientich'],
        tinh_id[0]
    ))
    vung_id = cur.fetchone()[0]
    
    # 3. Tạo vung_cay_trong
    cur.execute("""
        INSERT INTO nongsan.vung_cay_trong
        (vung_trong_id, loai_cay_id, dien_tich)
        VALUES (%s, %s, %s)
        RETURNING id
    """, (vung_id, row['caytrong_ID'], row['dientich']))
    vung_cay_id = cur.fetchone()[0]
    
    # 4. Parse và insert thị trường
    if pd.notna(row['thitruongxuatkhau']):
        thi_truongs = row['thitruongxuatkhau'].split(',')
        for tt_name in thi_truongs:
            tt_name = tt_name.strip()
            cur.execute("""
                SELECT id FROM nongsan.thi_truong 
                WHERE ten_thi_truong ILIKE %s
            """, (f'%{tt_name}%',))
            tt_id = cur.fetchone()
            
            if tt_id:
                cur.execute("""
                    INSERT INTO nongsan.vung_thi_truong
                    (vung_cay_trong_id, thi_truong_id)
                    VALUES (%s, %s)
                    ON CONFLICT DO NOTHING
                """, (vung_cay_id, tt_id[0]))

conn.commit()
```

## Script Import Hoàn Chỉnh

### Tạo file `import_data.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Script import dữ liệu từ Excel vào PostgreSQL
"""

import pandas as pd
import psycopg2
from psycopg2.extras import execute_batch
import os
from pathlib import Path

# Cấu hình database
DB_CONFIG = {
    'host': 'localhost',
    'database': 'nongsan_db',
    'user': 'postgres',
    'password': 'your_password'
}

# Thư mục dữ liệu
BASE_DIR = Path(__file__).parent

def get_connection():
    """Tạo connection đến PostgreSQL"""
    return psycopg2.connect(**DB_CONFIG)

def import_loai_cay():
    """Import loại cây từ gen_caygiong.xlsx"""
    print("Importing loại cây...")
    
    df = pd.read_excel(BASE_DIR / 'giong/gen_caygiong.xlsx', sheet_name='Sheet2')
    
    conn = get_connection()
    cur = conn.cursor()
    
    for _, row in df.iterrows():
        cur.execute("""
            INSERT INTO nongsan.loai_cay (ma_cay, ten_cay, ten_khoa_hoc)
            VALUES (%s, %s, %s)
            ON CONFLICT (ma_cay) DO NOTHING;
        """, (
            f"CAY_{row['caytrong_ID']:03d}",
            row['ten_cay_trong'],
            row['ten_khoa_hoc']
        ))
    
    conn.commit()
    print(f"  ✓ Imported {len(df)} loại cây")
    cur.close()
    conn.close()

def import_giong_bao_ho():
    """Import giống bảo hộ"""
    print("Importing giống bảo hộ...")
    
    df = pd.read_excel(BASE_DIR / 'giong/giong_baoho.xlsx')
    
    conn = get_connection()
    cur = conn.cursor()
    
    for _, row in df.iterrows():
        # Parse date
        ngay_hieu_luc = pd.to_datetime(row['ngaydk_bd_hieuluc'], errors='coerce')
        
        cur.execute("""
            INSERT INTO nongsan.giong_bao_ho 
            (so_bang, ma_giong, ten_giong, ten_chu_so_huu, 
             ngay_bat_dau_hieu_luc, tinh_trang)
            VALUES (%s, %s, %s, %s, %s, %s)
            ON CONFLICT (so_bang) DO NOTHING;
        """, (
            row['sobang'],
            row['giong_ID'],
            row['tengiong'],
            row['tenchusohuu'],
            ngay_hieu_luc,
            row['Tình trạng bằng']
        ))
    
    conn.commit()
    print(f"  ✓ Imported {len(df)} giống bảo hộ")
    cur.close()
    conn.close()

def import_msvt_vung_trong():
    """Import vùng trồng từ MSVT"""
    print("Importing vùng trồng (MSVT)...")
    
    df = pd.read_excel(BASE_DIR / 'msvt/msvt_thitruongvungtrong.xlsx')
    
    conn = get_connection()
    cur = conn.cursor()
    
    for idx, row in df.iterrows():
        try:
            # Tạo vùng trồng
            cur.execute("""
                INSERT INTO nongsan.vung_trong 
                (ma_vung, ma_vung_puc, ten_vung, dien_tich)
                VALUES (%s, %s, %s, %s)
                RETURNING id;
            """, (
                row['maso'] if pd.notna(row['maso']) else row['mavungtrong_puc'],
                row['mavungtrong_puc'],
                row['tenvungtrong'],
                row['dientich']
            ))
            vung_id = cur.fetchone()[0]
            
            # Commit mỗi vùng
            conn.commit()
            print(f"  ✓ Imported vùng {idx+1}/{len(df)}: {row['tenvungtrong']}")
            
        except Exception as e:
            conn.rollback()
            print(f"  ✗ Error at row {idx}: {e}")
    
    cur.close()
    conn.close()

def main():
    """Main function"""
    print("="*80)
    print("  IMPORT DỮ LIỆU TỪ EXCEL VÀO POSTGRESQL")
    print("="*80)
    print()
    
    try:
        # Test connection
        conn = get_connection()
        conn.close()
        print("✓ Kết nối database thành công\n")
        
        # Import từng module
        import_loai_cay()
        import_giong_bao_ho()
        import_msvt_vung_trong()
        # TODO: Thêm các hàm import khác
        
        print("\n" + "="*80)
        print("  ✓ HOÀN THÀNH IMPORT DỮ LIỆU")
        print("="*80)
        
    except Exception as e:
        print(f"\n✗ LỖI: {e}")
        return 1
    
    return 0

if __name__ == '__main__':
    exit(main())
```

## Lưu Ý Quan Trọng

### 1. Xử Lý Encoding
```python
# Nếu file Excel có encoding issues
df = pd.read_excel(file_path, encoding='utf-8-sig')
```

### 2. Xử Lý NULL/NaN
```python
# Kiểm tra NULL trước khi insert
if pd.notna(row['column_name']):
    value = row['column_name']
else:
    value = None
```

### 3. Xử Lý Date
```python
# Parse date an toàn
date_value = pd.to_datetime(row['date_column'], errors='coerce')
```

### 4. Transaction & Rollback
```python
try:
    # Insert nhiều dòng
    conn.commit()
except Exception as e:
    conn.rollback()
    print(f"Error: {e}")
```

### 5. Bulk Insert (Nhanh hơn)
```python
from psycopg2.extras import execute_batch

data = [(val1, val2, val3) for row in df.itertuples()]

execute_batch(cur, """
    INSERT INTO table (col1, col2, col3) VALUES (%s, %s, %s)
""", data, page_size=1000)
```
